#load packages:

import json
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go

### load data here:
gap = px.data.gapminder()

gap.head()

gap_1997 = gap.query("continent=='Europe'").query("year==2007")

fig = px.choropleth(
    gap_1997,
    locations = "iso_alpha",
    color = "gdpPercap",
    hover_name = "country",
    hover_data = {
        #"lifeExpectancy":":1f",
        "pop":":,",
        "gdpPercap":":,.0f",
        "iso_alpha":False
    },
    color_continuous_scale="Viridis",
    title = "GDP per capita by country",
    subtitle = "Europe, 2007",
)

fig.show()
fig.update_geos(projection_type="azimuthal equal area")

# I added 'azimuthal equal area' to show countries and regions exactly to their true relative sizes
# Choosing a year
gap = px.data.gapminder().query("year==2007")

# Creating bubble map
fig = px.scatter_geo(
    gap,
    locations="iso_alpha",
    size="pop",           
    color="continent",      
    hover_name="country",
    hover_data={
        "pop": True,
        "lifeExp": True,
        "continent": False,
        "iso_alpha": False
    },
    title="Life expectancy by country (2007)",
    projection="natural earth"    
    #I added  natural earth projection. This makes it look more natural than a flat rectangle
)

fig.show()
